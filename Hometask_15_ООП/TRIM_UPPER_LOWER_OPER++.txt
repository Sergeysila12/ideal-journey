class string

        int index = 0;
        for (; char_string[index] != '\0'; index++)
        {
        }
        this->size = index + 1;

        array = new char[this->size];

        for (int i = 0; i < size; i++)

            array[i] = char_string[i];

        for (int i = 0; i < size; i++)

    }
    string upper(char* arr)
    {
        int index = 0;
        for (; arr[index] != '\0'; index++) {}
        index++;
        for (int i = 0; arr[i] < index; i++)
        {
            if (arr[i] >= 0x61 && arr[i] <= 0x7A)
            {
                arr[i]-0x20;
            }
        }
        string up { arr, index };
        return up;
    }
    string lower(char* arr)
    {
        int index = 0;
        for (; arr[index] != '\0'; index++) {}
        index++;
        for (int i = 0; arr[i] < index; i++)
        {
            if (arr[i] >= 0x41 && arr[i] <= 0x5A)
            {
                arr[i] + 0x20;
            }
        }
        string low{ arr, index };
        std::cout << "lower!\n";

    }

    string trim(char* arr)
    {
        int index = 0;
        int essential = 0;
        int flag = 0;
        int mark = 0;
        for (; arr[index] != '\0'; index++) {  }
        index++;

        char* str = new char[index];

        for (int  j = 0, k=1; j < index; j++,k++)
        {
            flag = 0;
            mark = 0;
            if (arr[j] != 0x20 && arr[k] == 0x20)
            {
                for (int z = k; z < index; z++)
                {
                    if (arr[z] != 0x20)
                    {
                        mark = 1;
                    }
                }
                if (mark)
                {
                    flag = 1;
                }
                else
                {
                    str[essential++] = arr[j];
                    break;
                }
   
            }

            {
               flag=1;
            }
           if (flag)
           {
               str[essential++] = arr[j];
           }
        }
        str[essential] = '\0';
        string trimmer{ str, essential };
        delete[] str;
        return trimmer;
    }
