Создайте класс Device (например, имитация графического устройства).

Избежание циклических ссылок с weak_ptr
Создайте два класса: Parent и Child. Parent содержит std::shared_ptr<Child>, а Child хранит std::weak_ptr<Parent>. 
Убедитесь, что при удалении Parent объект Child корректно освобождается.

#include <iostream>
#include <memory>
#include <stdexcept>

template <typename T>
class Smart_ptr
{
protected:
    T* ptr;
public:
	Smart_ptr(T value)
	{
		ptr = new T(value);
	}
	Smart_ptr(T* memory)
	{
		ptr = memory;
	}
	~Smart_ptr()
	{
		delete ptr;
	}
	T& operator*()
	{
		return *ptr;
	}
};

void foo()
{
	Smart_ptr<int> pt(0);
}
