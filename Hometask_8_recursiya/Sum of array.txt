
#include <iostream>

//написать рекурсивную функцию, 
// которая примет массив и найдёт сумму его элементов

int FindSum(int arr[], int SIZE)
{
    if (SIZE == 0)
  {
       return 0;
   }
   return arr[SIZE - 1]+ FindSum(arr, SIZE -1);
}


   
int main()
{
    setlocale(LC_ALL, "ru");
    srand(time(NULL));
    int const SIZE = 4;
    int arr[SIZE];
    std::cout << "Массив: " << std::endl;
    for (int i = 0; i < SIZE; ++i) {
        arr[i] = rand()%101;
      std::cout << arr[i] << "\t";
    }
    std::cout << std::endl<<"Сумма элементов в массиве: \n" << FindSum( arr,  SIZE);

}
